🔹 Overview of the Rendering Pipeline
1️⃣ Initialize OpenGL & wxGLCanvas
2️⃣ Create a Shader Program (to map uint8_t values to colors via a LUT)
3️⃣ Load & Update Data Efficiently (new pings appear on the right, scrolling left)
4️⃣ Use a Texture for Fast Rendering (Avoid slow per-pixel processing)
5️⃣ Optimize Performance (VBOs, double buffering, efficient memory usage)

EchogramCanvas::EchogramCanvas(wxFrame* parent)
    : wxGLCanvas(parent, wxID_ANY, nullptr) {
    m_context = new wxGLContext(this);
    SetCurrent(*m_context);
    
    glewExperimental = GL_TRUE;
    if (glewInit() != GLEW_OK) {
        wxMessageBox("Failed to initialize GLEW!", "Error", wxOK | wxICON_ERROR);
        return;
    }

    InitGL();
    CreateShaders();
    CreateTexture();
}

void EchogramCanvas::InitGL() {
    glEnable(GL_TEXTURE_2D);
    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
}

const char* vertexShaderSrc = R"(
    #version 330 core
    layout (location = 0) in vec2 aPos;
    layout (location = 1) in vec2 aTexCoord;
    out vec2 TexCoord;
    void main() {
        TexCoord = aTexCoord;
        gl_Position = vec4(aPos, 0.0, 1.0);
    }
)";

const char* fragmentShaderSrc = R"(
    #version 330 core
    in vec2 TexCoord;
    out vec4 FragColor;
    uniform sampler2D echogramTexture;
    void main() {
        FragColor = texture(echogramTexture, TexCoord);
    }
)";

void EchogramCanvas::CreateShaders() {
    GLuint vertexShader = glCreateShader(GL_VERTEX_SHADER);
    glShaderSource(vertexShader, 1, &vertexShaderSrc, nullptr);
    glCompileShader(vertexShader);

    GLuint fragmentShader = glCreateShader(GL_FRAGMENT_SHADER);
    glShaderSource(fragmentShader, 1, &fragmentShaderSrc, nullptr);
    glCompileShader(fragmentShader);

    shaderProgram = glCreateProgram();
    glAttachShader(shaderProgram, vertexShader);
    glAttachShader(shaderProgram, fragmentShader);
    glLinkProgram(shaderProgram);

    glDeleteShader(vertexShader);
    glDeleteShader(fragmentShader);
}

void EchogramCanvas::CreateTexture() {
    glGenTextures(1, &textureID);
    glBindTexture(GL_TEXTURE_2D, textureID);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);

    echogramData.resize(1024 * 256, 0);  // Initialize empty echogram (1024 bins x 256 lines)
    glTexImage2D(GL_TEXTURE_2D, 0, GL_RED, 1024, 256, 0, GL_RED, GL_UNSIGNED_BYTE, echogramData.data());
}

void EchogramCanvas::ScrollData(const std::vector<uint8_t>& newPing) {
    // Shift left by 1 column (each row moves left)
    for (int row = 0; row < 256; ++row) {
        for (int col = 0; col < 1023; ++col) {
            echogramData[row * 1024 + col] = echogramData[row * 1024 + col + 1];
        }
        echogramData[row * 1024 + 1023] = newPing[row];  // Add new ping on the right
    }

    // Update OpenGL texture
    glBindTexture(GL_TEXTURE_2D, textureID);
    glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, 1024, 256, GL_RED, GL_UNSIGNED_BYTE, echogramData.data());
}

void EchogramCanvas::Render() {
    glClear(GL_COLOR_BUFFER_BIT);
    
    glUseProgram(shaderProgram);
    glBindTexture(GL_TEXTURE_2D, textureID);

    glBegin(GL_QUADS);
        glTexCoord2f(0.0f, 0.0f); glVertex2f(-1.0f, -1.0f);
        glTexCoord2f(1.0f, 0.0f); glVertex2f(1.0f, -1.0f);
        glTexCoord2f(1.0f, 1.0f); glVertex2f(1.0f, 1.0f);
        glTexCoord2f(0.0f, 1.0f); glVertex2f(-1.0f, 1.0f);
    glEnd();

    SwapBuffers();
}

void EchogramCanvas::OnPaint(wxPaintEvent& event) {
    wxPaintDC dc(this);
    Render();
}
